CREATE YOU PROJECT FOLDER -

mkdir scripts

PART 2 : AUTHENTICATION COGNITO FOR ACCESS

1. Create the user pool and store the pool id in env file

aws cognito-idp create-user-pool --region $REGION --pool-name $PREFIX-pool \
  --admin-create-user-config AllowAdminCreateUserOnly=true --query UserPool.Id --output text

2. Create user pool client and add Client id in env file

aws cognito-idp create-user-pool-client --region $REGION --user-pool-id $POOL_ID \
  --client-name $PREFIX-cli --explicit-auth-flows ALLOW_USER_PASSWORD_AUTH ALLOW_REFRESH_TOKEN_AUTH \
  --no-generate-secret --query UserPoolClient.ClientId --output text

3. Add DISCOVERY_URL in th env file 
https://cognito-idp.${REGION}.amazonaws.com/${POOL_ID}/.well-known/openid-configuration


4. Create 3 test users - 
Replace <user1> for each user or run in a loop

aws cognito-idp admin-create-user --region $REGION --user-pool-id $POOL_ID --username "user1" --message-action SUPPRESS --query User.Username --output text
aws cognito-idp admin-set-user-password --region $REGION --user-pool-id $POOL_ID --username "user1" --password "$TEST_PASSWORD" --permanent



for u in "$TEST_USER1" "$TEST_USER2" "$TEST_USER3"; do
  aws cognito-idp admin-create-user --region $REGION --user-pool-id $POOL_ID --username "$u" --message-action SUPPRESS --query User.Username --output text
  aws cognito-idp admin-set-user-password --region $REGION --user-pool-id $POOL_ID --username "$u" --password "$TEST_PASSWORD" --permanent
done
